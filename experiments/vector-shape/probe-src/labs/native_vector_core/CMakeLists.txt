cmake_minimum_required(VERSION 3.24)
project(CNLNativeVectorCore LANGUAGES CXX)
add_library(CNLNativeVectorCore SHARED CNLNativeVectorCore.cpp CNLNativeVectorCore.h)
target_compile_features(CNLNativeVectorCore PRIVATE cxx_std_17)
target_compile_definitions(CNLNativeVectorCore PRIVATE CNL_VECTOR_CORE_EXPORTS)
set_target_properties(CNLNativeVectorCore PROPERTIES PREFIX "" OUTPUT_NAME "CNLNativeVectorCore")
add_executable(NativeVectorCoreTest NativeVectorCoreTest.cpp)
target_compile_features(NativeVectorCoreTest PRIVATE cxx_std_17)
target_link_libraries(NativeVectorCoreTest PRIVATE CNLNativeVectorCore)
enable_testing()
add_test(NAME native-vector-core COMMAND NativeVectorCoreTest)
